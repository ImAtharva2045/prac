const express = require("express");
const Student = require("../models/Student");

const router = express.Router();

// GET ALL STUDENTS
router.get("/", async (req, res) => {
    try {
        const students = await Student.find();
        res.json(students);
    } catch (error) {
        res.status(500).json({
            message: "Error getting students"
        });
    }
});

// GET ONE STUDENT
router.get("/:id", async (req, res) => {
    try {
        const student = await Student.findById(req.params.id);
        res.json(student);
    } catch (error) {
        res.status(500).json({
            message: "Student not found"
        });
    }
});

// ADD STUDENT
router.post("/", async (req, res) => {
    try {
        const student = new Student({
            name: req.body.name,
            rollNo: req.body.rollNo,
            branch: req.body.branch,
            marks: req.body.marks,
            attendance: req.body.attendance
        });

        const savedStudent = await student.save();

        res.json(savedStudent);
    } catch (error) {
        res.status(500).json({
            message: "Error adding student"
        });
    }
});

// UPDATE STUDENT
router.put("/:id", async (req, res) => {
    try {
        const student = await Student.findByIdAndUpdate(
            req.params.id,
            {
                name: req.body.name,
                rollNo: req.body.rollNo,
                branch: req.body.branch,
                marks: req.body.marks,
                attendance: req.body.attendance
            },
            { new: true }
        );

        res.json(student);
    } catch (error) {
        res.status(500).json({
            message: "Error updating student"
        });
    }
});

// DELETE STUDENT
router.delete("/:id", async (req, res) => {
    try {
        await Student.findByIdAndDelete(req.params.id);

        res.json({
            message: "Student deleted"
        });
    } catch (error) {
        res.status(500).json({
            message: "Error deleting student"
        });
    }
});

module.exports = router;

import { useEffect, useState } from "react";
import axios from "axios";

function Teacher() {
    const [students, setStudents] = useState([]);

    // Form states
    const [name, setName] = useState("");
    const [rollNo, setRollNo] = useState("");
    const [branch, setBranch] = useState("");
    const [marks, setMarks] = useState("");
    const [attendance, setAttendance] = useState("");

    // null means Add mode, otherwise Edit mode
    const [editId, setEditId] = useState(null);

    // GET ALL STUDENTS
    const getStudents = async () => {
        const response = await axios.get(
            "http://localhost:5000/api/students"
        );

        setStudents(response.data);
    };

    // Run when page opens
    useEffect(() => {
        getStudents();
    }, []);

    // ADD OR UPDATE STUDENT
    const handleSubmit = async (e) => {
        e.preventDefault();

        // Basic validation
        if (
            name === "" ||
            rollNo === "" ||
            branch === "" ||
            marks === "" ||
            attendance === ""
        ) {
            alert("Please fill all fields");
            return;
        }

        const student = {
            name: name,
            rollNo: Number(rollNo),
            branch: branch,
            marks: Number(marks),
            attendance: Number(attendance)
        };

        // ADD STUDENT
        if (editId === null) {
            await axios.post(
                "http://localhost:5000/api/students",
                student
            );

            alert("Student added");
        }

        // UPDATE STUDENT
        else {
            await axios.put(
                `http://localhost:5000/api/students/${editId}`,
                student
            );

            alert("Student updated");
            setEditId(null);
        }

        // Clear form
        setName("");
        setRollNo("");
        setBranch("");
        setMarks("");
        setAttendance("");

        // Refresh student list
        getStudents();
    };

    // EDIT STUDENT
    const editStudent = (student) => {
        setEditId(student._id);
        setName(student.name);
        setRollNo(student.rollNo);
        setBranch(student.branch);
        setMarks(student.marks);
        setAttendance(student.attendance);
    };

    // DELETE STUDENT
    const deleteStudent = async (id) => {
        await axios.delete(
            `http://localhost:5000/api/students/${id}`
        );

        alert("Student deleted");

        getStudents();
    };

    return (
        <div>
            <h1>Teacher Dashboard</h1>

            <h2>
                {editId === null ? "Add Student" : "Edit Student"}
            </h2>

            <form onSubmit={handleSubmit}>
                <input
                    type="text"
                    placeholder="Name"
                    value={name}
                    onChange={(e) => setName(e.target.value)}
                />

                <br />
                <br />

                <input
                    type="number"
                    placeholder="Roll No"
                    value={rollNo}
                    onChange={(e) => setRollNo(e.target.value)}
                />

                <br />
                <br />

                <input
                    type="text"
                    placeholder="Branch"
                    value={branch}
                    onChange={(e) => setBranch(e.target.value)}
                />

                <br />
                <br />

                <input
                    type="number"
                    placeholder="Marks"
                    value={marks}
                    onChange={(e) => setMarks(e.target.value)}
                />

                <br />
                <br />

                <input
                    type="number"
                    placeholder="Attendance"
                    value={attendance}
                    onChange={(e) => setAttendance(e.target.value)}
                />

                <br />
                <br />

                <button type="submit">
                    {editId === null ? "Add Student" : "Update Student"}
                </button>
            </form>

            <hr />

            <h2>All Students</h2>

            {students.map((student) => (
                <div key={student._id}>
                    <p>Name: {student.name}</p>
                    <p>Roll No: {student.rollNo}</p>
                    <p>Branch: {student.branch}</p>
                    <p>Marks: {student.marks}</p>
                    <p>Attendance: {student.attendance}%</p>

                    <button
                        onClick={() => editStudent(student)}
                    >
                        Edit
                    </button>

                    {" "}

                    <button
                        onClick={() => deleteStudent(student._id)}
                    >
                        Delete
                    </button>

                    <hr />
                </div>
            ))}
        </div>
    );
}

export default Teacher;

        if (id === "") {
            alert("Enter student ID");
            return;
        }

        try {
            const response = await axios.get(
                `http://localhost:5000/api/students/${id}`
            );

            setStudent(response.data);
        } catch (error) {
            alert("Student not found");
        }
    };

    return (
        <div>
            <h1>Student Dashboard</h1>

            <input
                type="text"
                placeholder="Enter Student ID"
                value={id}
                onChange={(e) => setId(e.target.value)}
            />

            <button onClick={getStudent}>
                View Performance
            </button>

            {student && (
                <div>
                    <h2>Performance Report</h2>

                    <p>Name: {student.name}</p>
                    <p>Roll No: {student.rollNo}</p>
                    <p>Branch: {student.branch}</p>
                    <p>Marks: {student.marks}</p>
                    <p>Attendance: {student.attendance}%</p>
                </div>
            )}
        </div>
    );
}

export default Student;

import { useEffect, useState } from "react";
import axios from "axios";

function User() {
    const [books, setBooks] = useState([]);
    const [search, setSearch] = useState("");

    // GET ALL BOOKS
    const getBooks = async () => {
        try {
            const response = await axios.get(
                "http://localhost:5000/api/books"
            );

            setBooks(response.data);
        } catch (error) {
            console.log(error);
        }
    };

    // RUN WHEN PAGE OPENS
    useEffect(() => {
        getBooks();
    }, []);

    // SEARCH BOOKS
    const filteredBooks = books.filter((book) => {
        return book.title
            .toLowerCase()
            .includes(search.toLowerCase());
    });

    // PURCHASE BOOK
    const purchaseBook = async (book) => {
        const order = {
            bookId: book._id,
            bookTitle: book.title,
            quantity: 1
        };

        await axios.post(
            "http://localhost:5000/api/orders",
            order
        );

        alert("Book purchased");
    };

    return (
        <div>
            <h1>User Dashboard</h1>

            <h2>Search Books</h2>

            <input
                type="text"
                placeholder="Search book"
                value={search}
                onChange={(e) => setSearch(e.target.value)}
            />

            <br />
            <br />

            <h2>Available Books</h2>

            {filteredBooks.map((book) => (
                <div key={book._id}>
                    <p>Title: {book.title}</p>
                    <p>Author: {book.author}</p>
                    <p>Price: {book.price}</p>

                    <button
                        onClick={() => purchaseBook(book)}
                    >
                        Purchase
                    </button>

                    <hr />
                </div>
            ))}
        </div>
    );
}

export default User;
